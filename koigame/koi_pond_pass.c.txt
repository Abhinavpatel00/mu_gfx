/*
   Integration sketch for mu_gfx.

   The exact pipeline helper calls should use the existing engine helpers;
   this file deliberately avoids inventing a second Vulkan abstraction.
*/

/* Renderer state: fixed capacity, one GPU buffer per frame-in-flight. */

typedef struct KoiPondFrame {
    Buffer buffer;
    KoiPondData *mapped;
} KoiPondFrame;

/* Engine state additions: */
KoiGame game;
KoiPondFrame koi_frame[MAX_FRAMES_IN_FLIGHT];
uint32_t koi_pond_pipeline;

/* Per-frame CPU work: */
void koi_update_frame(Renderer *r, float dt)
{
    static const float fixed_dt = 1.0f / 60.0f;
    static float accumulator;

    accumulator += dt;

    // Fixed simulation time makes gameplay deterministic and decouples it
    // from render rate. Cap catch-up so a breakpoint cannot cause a huge burst.
    uint32_t steps = 0;
    while (accumulator >= fixed_dt && steps < 4) {
        koi_game_update(&game, fixed_dt);
        accumulator -= fixed_dt;
        ++steps;
    }

    KoiPondData *gpu = koi_frame[r->current_frame].mapped;
    koi_game_write_gpu(&game, gpu);
}

/* Input translation: */
void koi_pointer_down(float x, float y, bool right_button)
{
    // Pond coordinates match the shader: x in [0,3], y in [0,1].
    KoiGameVec2 p = { x * 3.0f, 1.0f - y };
    koi_game_input(&game,
                   right_button ? KOI_INPUT_STIR : KOI_INPUT_FEED,
                   p);
}

/* Graphics pass: */
static void pass_koi_pond(Renderer *r, VkCommandBuffer cmd)
{
    uint32_t image = r->swapchain.current_image;
    GpuProfiler *frame_prof = &r->gpuprofiler[r->current_frame];

    GPU_SCOPE(frame_prof, cmd, "Koi Pond", VK_PIPELINE_STAGE_2_COLOR_ATTACHMENT_OUTPUT_BIT) {
        /*
           1. Transition hdr_color[image] exactly like pass_fire.
           2. Begin dynamic rendering with hdr_color as COLOR_ATTACHMENT.
           3. Bind koi_pond_pipeline.
           4. Bind the already-global Set 0 once for the frame.
           5. Draw the fullscreen triangle: vkCmdDraw(cmd, 3, 1, 0, 0).
           6. End rendering.

           No descriptor bind belongs in the fish/ripple loop because the pond
           buffer is a frame-global resource, not a per-object binding.
        */
    }
}

/* Frame order: */
pass_koi_pond(r, cmd);
post_pass(r, cmd);
pass_smaa(r, cmd);
pass_ldr_to_swapchain(r, cmd);
